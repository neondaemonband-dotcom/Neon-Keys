cmake_minimum_required(VERSION 3.22)
project(NeonKeys VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT DEFINED JUCE_DIR)
  message(FATAL_ERROR "Pass -DJUCE_DIR=path/to/JUCE")
endif()
add_subdirectory("${JUCE_DIR}" JUCE)
juce_add_plugin(NeonKeys COMPANY_NAME "Neon Daemon" BUNDLE_ID "com.neondaemon.neonkeys" PLUGIN_MANUFACTURER_CODE Ndmn PLUGIN_CODE Nk01 FORMATS VST3 PRODUCT_NAME "Neon Keys" IS_SYNTH TRUE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
target_sources(NeonKeys PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp)
juce_generate_juce_header(NeonKeys)
target_compile_definitions(NeonKeys PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(NeonKeys PRIVATE juce::juce_audio_utils juce::juce_dsp juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)
